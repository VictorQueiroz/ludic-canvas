cmake_minimum_required(VERSION 3.28)
project(LudicNative LANGUAGES C CXX OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_OSX_DEPLOYMENT_TARGET 26.0)
if(NOT APPLE)
  message(FATAL_ERROR "This proof host currently targets macOS. The core and layout interface are portable.")
endif()
get_filename_component(LC_ROOT "../.." ABSOLUTE)
set(LC_CACHE "${LC_ROOT}/.toolchain/native" CACHE PATH "Native dependency cache")
file(READ "${LC_ROOT}/tools/native/dependencies.json" LC_DEPENDENCIES)
string(JSON HERMES_REVISION GET "${LC_DEPENDENCIES}" hermes revision)
string(JSON SKIA_REVISION GET "${LC_DEPENDENCIES}" skia revision)
string(JSON YOGA_REVISION GET "${LC_DEPENDENCIES}" yoga revision)
set(HERMES_SOURCE "${LC_CACHE}/hermes-${HERMES_REVISION}")
set(HERMES_BUILD "${LC_CACHE}/hermes-build")
set(SKIA_SOURCE "${LC_CACHE}/skia-${SKIA_REVISION}")
set(YOGA_SOURCE "${LC_CACHE}/yoga-${YOGA_REVISION}")
file(GLOB_RECURSE LC_YOGA_SOURCES CONFIGURE_DEPENDS "${YOGA_SOURCE}/yoga/*.cpp")
add_library(ludic_yoga STATIC ${LC_YOGA_SOURCES})
target_include_directories(ludic_yoga PUBLIC "${YOGA_SOURCE}")
add_library(ludic_engine STATIC engine.cpp)
target_include_directories(ludic_engine PUBLIC "${HERMES_SOURCE}/API" "${HERMES_SOURCE}/API/jsi" "${HERMES_SOURCE}/public" "${HERMES_BUILD}/public" "${HERMES_BUILD}")
target_link_libraries(ludic_engine PUBLIC ludic_yoga "${HERMES_BUILD}/lib/libhermesvm_a.a" "${HERMES_BUILD}/jsi/libjsi.a" "${HERMES_BUILD}/external/boost/boost_1_86_0/libs/context/libboost_context.a")
target_link_libraries(ludic_engine PUBLIC "-framework CoreFoundation")
add_executable(ludic-contract contract.cpp)
target_link_libraries(ludic-contract PRIVATE ludic_engine)
add_executable(ludic-native MACOSX_BUNDLE macos.mm renderer.cpp)
add_executable(ludic-host-test host-test.mm macos.mm renderer.cpp)
target_compile_definitions(ludic-host-test PRIVATE LUDIC_HOST_TEST)
add_executable(ludic-renderer-test renderer-test.cpp renderer.cpp)
target_include_directories(ludic-renderer-test PRIVATE "${SKIA_SOURCE}")
target_compile_definitions(ludic-renderer-test PRIVATE SK_GANESH SK_METAL)
target_link_libraries(ludic-renderer-test PRIVATE "${SKIA_SOURCE}/out/ludic/libskia.a" "${SKIA_SOURCE}/out/ludic/libskcms.a" "-framework Metal" "-framework Foundation" "-framework CoreGraphics" "-framework CoreText")
foreach(target ludic-native ludic-host-test)
  target_include_directories(${target} PRIVATE "${SKIA_SOURCE}")
  target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
  target_compile_definitions(${target} PRIVATE SK_GANESH SK_METAL)
  target_link_libraries(${target} PRIVATE ludic_engine "${SKIA_SOURCE}/out/ludic/libskia.a" "${SKIA_SOURCE}/out/ludic/libskcms.a" "-framework AppKit" "-framework Metal" "-framework QuartzCore" "-framework CoreGraphics" "-framework CoreText" "-framework ImageIO" "-framework Foundation")
endforeach()
set_target_properties(ludic-native PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "dev.ludiccanvas.portable-proof" MACOSX_BUNDLE_BUNDLE_NAME "Ludic Native" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist")
